import re

class TokenizadorSistemasExpertos:
    def __init__(self):
        self.tabla_atomos = {}
        self.letras_disponibles = ['p', 'q', 'r', 's', 't', 'u', 'v', 'w', 'x', 'y', 'z']
        
        self.palabras_clave = {
            'y': 'CONJ',
            'pero': 'CONJ',
            'o': 'DISJ',
            'no': 'NEG',
            'si': 'COND_SI',
            'entonces': 'COND_ENTONCES',
            'si y solo si': 'BICOND'
        }

    def limpiar_texto(self, texto):
        """Limpia puntuación, pero conserva una marca para la coma condicional si es necesario."""
        texto = texto.lower().strip()
        # Transforma la primera coma en un 'entonces' implícito si la frase empieza con 'si'
        if texto.startswith("si ") and " entonces " not in texto and "," in texto:
            texto = texto.replace(",", " entonces", 1)
        
        texto = re.sub(r'[.;:]', '', texto)
        texto = texto.replace(',', '') # Quita comas sobrantes
        texto = re.sub(r'\s+', ' ', texto)
        return texto

    def segmentar_palabras(self, texto):
        texto = self.limpiar_texto(texto)
        texto = re.sub(r'\bsi\s+y\s+solo\s+si\b', '##BICOND##', texto)
        
        tokens_preliminares = texto.split()
        tokens_finales = []
        
        for t in tokens_preliminares:
            if t == '##BICOND##':
                tokens_finales.append('si y solo si')
            else:
                tokens_finales.append(t)
        return tokens_finales

    def registrar_atomos_de_base(self, lineas):
        for linea in lineas:
            palabras = self.segmentar_palabras(linea)
            if not palabras:
                continue
            
            buffer_atomo = []
            for palabra in palabras:
                if palabra in self.palabras_clave:
                    if buffer_atomo:
                        self._guardar_atomo_puro(buffer_atomo)
                        buffer_atomo = []
                else:
                    buffer_atomo.append(palabra)
            if buffer_atomo:
                self._guardar_atomo_puro(buffer_atomo)

    def _guardar_atomo_puro(self, lista_palabras):
        if lista_palabras and lista_palabras[0] == 'no' and len(lista_palabras) > 1:
            texto_atomo = " ".join(lista_palabras[1:])
        else:
            texto_atomo = " ".join(lista_palabras)
            
        if texto_atomo and texto_atomo not in self.tabla_atomos and self.letras_disponibles:
            self.tabla_atomos[texto_atomo] = self.letras_disponibles.pop(0)

    def buscar_letra_atomo(self, lista_palabras):
        if not lista_palabras:
            return ""
        
        if lista_palabras[0] == 'no' and len(lista_palabras) > 1:
            texto = " ".join(lista_palabras[1:])
            es_negado = True
        else:
            texto = " ".join(lista_palabras)
            es_negado = False
            
        letra = self.tabla_atomos.get(texto, "")
        if not letra:
            return ""
        return f"¬{letra}" if es_negado else letra

    def traducir_oracion(self, linea_texto):
        palabras = self.segmentar_palabras(linea_texto)
        if not palabras:
            return ""

        es_condicional_estandar = (palabras[0] == 'si') if palabras else False

        componentes_formula = []
        buffer_atomo = []
        
        i = 0
        while i < len(palabras):
            palabra = palabras[i]
            
            if palabra in self.palabras_clave:
                if buffer_atomo:
                    letra = self.buscar_letra_atomo(buffer_atomo)
                    if letra: componentes_formula.append(letra)
                    buffer_atomo = []
                
                tipo_conector = self.palabras_clave[palabra]
                
                if tipo_conector == 'CONJ': componentes_formula.append('∧')
                elif tipo_conector == 'DISJ': componentes_formula.append('∨')
                elif tipo_conector == 'BICOND': componentes_formula.append('↔')
                elif tipo_conector == 'COND_ENTONCES': componentes_formula.append('→')
                elif tipo_conector == 'COND_SI' and i > 0: componentes_formula.append('←')
                elif tipo_conector == 'NEG':
                    buffer_atomo.append(palabra)
            else:
                buffer_atomo.append(palabra)
            i += 1

        if buffer_atomo:
            letra = self.buscar_letra_atomo(buffer_atomo)
            if letra: componentes_formula.append(letra)

        if componentes_formula and componentes_formula[0] == 'si':
            componentes_formula.pop(0)

        if '←' in componentes_formula:
            idx_flecha = componentes_formula.index('←')
            consecuencia = componentes_formula[:idx_flecha]
            condicion = componentes_formula[idx_flecha+1:]
            formula_final = f"({''.join(condicion)}→{''.join(consecuencia)})"
        else:
            formula_final = "".join(componentes_formula)
            if es_condicional_estandar and '→' in formula_final:
                formula_final = f"({formula_final})"

        return formula_final

    def procesar(self, lineas_usuario):
        self.registrar_atomos_de_base(lineas_usuario)
        
        print("\n=== TABLA DE ÁTOMOS DETECTADOS ===")
        for frase, letra in self.tabla_atomos.items():
            print(f"{letra} = \"{frase}\"")
        print("==================================\n")

        print("=== BASE DE CONOCIMIENTO TRADUCIDA ===")
        for idx, linea in enumerate(lineas_usuario, 1):
            formula = self.traducir_oracion(linea)
            print(f"R{idx}: {linea.strip()}  -->  {formula}")
        print("======================================")

class MotorInferencia:
    def __init__(self):
        self.analizador = AnalizadorFormulas()
        self.modelos_encontrados = []

    def construir_arbol(self, nodo):
        """
        Algoritmo recursivo que expande el árbol semántico aplicando reglas
        de inferencia moleculares (Alfa y Beta).
        """
        # 1. Verificar si hay contradicciones inmediatas en la rama actual
        if nodo.verificar_contradiccion():
            return nodo

        # 2. Filtrar y limpiar fórmulas que ya son puramente literales (átomos)
        formulas_restantes = []
        for f in nodo.formulas:
            tipo, izq, der = self.analizador.encontrar_conector_principal(f)
            if tipo == 'ATOMO':
                nodo.atomos_afirmados.add(izq)
            elif tipo == 'ATOMO_NEGADO':
                # Al ser ¬p, el átomo puro es el segundo carácter [1]
                nodo.atomos_negados.add(izq[1])
            else:
                formulas_restantes.append(f)

        # Volver a verificar contradicciones tras extraer nuevos literales literales
        if nodo.verificar_contradiccion():
            return nodo

        # 3. Criterio de Parada: Si no quedan fórmulas compuestas por aplicar, la rama tiene éxito
        if not formulas_restantes:
            return nodo

        # 4. Estrategia del Video: Buscar primero una regla que NO bifurque (Regla Alfa)
        formula_elegida = None
        tipo_elegido = None
        sub_izq, sub_der = None, None

        for f in formulas_restantes:
            tipo, izq, der = self.analizador.encontrar_conector_principal(f)
            # Reglas Alfa (No bifurcan)
            if tipo in ['CONJUNCION']:
                formula_elegida = f
                tipo_elegido = tipo
                sub_izq, sub_der = izq, der
                break
            elif tipo == 'NEGACION':
                # Evaluar qué hay dentro de la negación ¬(...)
                tipo_interno, int_izq, int_der = self.analizador.encontrar_conector_principal(izq)
                if tipo_interno in ['DISJUNCION', 'CONDICIONAL']:  # ¬(A∨B) o ¬(A→B) son Alfa
                    formula_elegida = f
                    tipo_elegido = f"NEG_{tipo_interno}"
                    sub_izq, sub_der = int_izq, int_der
                    break

        # 5. Si no se encontró ninguna Regla Alfa, tomamos la primera Regla Beta disponible
        if not formula_elegida:
            formula_elegida = formulas_restantes[0]
            tipo_elegido, sub_izq, sub_der = self.analizador.encontrar_conector_principal(formula_elegida)

        # Crear una lista de las fórmulas que quedarán pendientes (marcando como aplicada la elegida)
        pendientes_siguientes = [f for f in formulas_restantes if f != formula_elegida]

        # 6. APLICACIÓN DE REGLAS DE EXPANSIÓN
        # --- CASOS REGLAS ALFA (1 Rama) ---
        if tipo_elegido == 'CONJUNCION':
            # A ∧ B  --> Añadir A y B directamente abajo
            pendientes_siguientes.extend([sub_izq, sub_der])
            nodo.izquierdo = NodoArbolSemantico(pendientes_siguientes, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)

        elif tipo_elegido == 'NEG_DISJUNCION':
            # ¬(A ∨ B)  --> Es equivalente a: ¬A y ¬B hacia abajo
            pendientes_siguientes.extend([f"¬{sub_izq}", f"¬{sub_der}"])
            nodo.izquierdo = NodoArbolSemantico(pendientes_siguientes, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)

        elif tipo_elegido == 'NEG_CONDICIONAL':
            # ¬(A → B)  --> Significa que el antecedente ocurre pero el consecuente no: A y ¬B
            pendientes_siguientes.extend([sub_izq, f"¬{sub_der}"])
            nodo.izquierdo = NodoArbolSemantico(pendientes_siguientes, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)

        # --- CASOS REGLAS BETA (Bifurcación en 2 Ramas) ---
        elif tipo_elegido == 'DISJUNCION':
            # A ∨ B  --> Abre rama izquierda para A, rama derecha para B
            izq_formulas = pendientes_siguientes + [sub_izq]
            der_formulas = pendientes_siguientes + [sub_der]
            nodo.izquierdo = NodoArbolSemantico(izq_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            nodo.derecho = NodoArbolSemantico(der_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)
            self.construir_arbol(nodo.derecho)

        elif tipo_elegido == 'CONDICIONAL':
            # A → B  --> Equivale a: Rama izquierda ¬A, Rama derecha B
            izq_formulas = pendientes_siguientes + [f"¬{sub_izq}"]
            der_formulas = pendientes_siguientes + [sub_der]
            nodo.izquierdo = NodoArbolSemantico(izq_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            nodo.derecho = NodoArbolSemantico(der_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)
            self.construir_arbol(nodo.derecho)

        elif tipo_elegido == 'BICONDICIONAL':
            # A ↔ B  --> Ambos verdaderos (A y B) o ambos falsos (¬A y ¬B)
            izq_formulas = pendientes_siguientes + [sub_izq, sub_der]
            der_formulas = pendientes_siguientes + [f"¬{sub_izq}", f"¬{sub_der}"]
            nodo.izquierdo = NodoArbolSemantico(izq_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            nodo.derecho = NodoArbolSemantico(der_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
            self.construir_arbol(nodo.izquierdo)
            self.construir_arbol(nodo.derecho)

        elif tipo_elegido == 'NEGACION':
            # Doble negación: ¬(¬A) --> Se elimina y queda A
            if sub_izq.startswith('¬'):
                pendientes_siguientes.append(sub_izq[1:])
                nodo.izquierdo = NodoArbolSemantico(pendientes_siguientes, nodo.atomos_afirmados, nodo.atomos_negados)
                self.construir_arbol(nodo.izquierdo)
            else:
                # Negación de Conjunción: ¬(A ∧ B) --> Abre rama ¬A o rama ¬B
                tipo_int, int_izq, int_der = self.analizador.encontrar_conector_principal(sub_izq)
                if tipo_int == 'CONJUNCION':
                    izq_formulas = pendientes_siguientes + [f"¬{int_izq}"]
                    der_formulas = pendientes_siguientes + [f"¬{int_der}"]
                    nodo.izquierdo = NodoArbolSemantico(izq_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
                    nodo.derecho = NodoArbolSemantico(der_formulas, nodo.atomos_afirmados, nodo.atomos_negados)
                    self.construir_arbol(nodo.izquierdo)
                    self.construir_arbol(nodo.derecho)

        # Si tras procesar sus hijos, ambas ramas descendientes se cierran, esta rama también se considera cerrada
        if nodo.izquierdo and nodo.izquierdo.cerrada:
            if not nodo.derecho or nodo.derecho.cerrada:
                nodo.cerrada = True
                
        return nodo

    def recolectar_modelos(self, nodo):
        """Recorre el árbol buscando los caminos que se mantuvieron abiertos al finalizar."""
        if not nodo:
            return
        
        # Si es un nodo hoja (fin de un camino)
        if not nodo.izquierdo and not nodo.derecho:
            if not nodo.cerrada:
                modelo = {
                    'afirmados': list(nodo.atomos_afirmados),
                    'negados': list(nodo.atomos_negados)
                }
                # Evitar registrar modelos duplicados en el reporte final
                if modelo not in self.modelos_encontrados:
                    self.modelos_encontrados.append(modelo)
            return

        self.recolectar_modelos(nodo.izquierdo)
        self.recolectar_modelos(nodo.derecho)

    def imprimir_reporte(self, nodo_raiz):
        """Muestra de forma clara la consistencia o inconsistencia de la Base de Conocimiento."""
        self.modelos_encontrados = []
        self.recolectar_modelos(nodo_raiz)
        
        print("\n=== REPORTE DE MODELOS (MUNDOS POSIBLES) ===")
        if not self.modelos_encontrados:
            print("ALERTAS DEL SISTEMA EXPERTO:")
            print("❌ LA BASE DE CONOCIMIENTO ES INCONSISTENTE.")
            print("Todas las ramas contienen contradicciones y se han cerrado (✕).")
        else:
            print(f"✅ LA BASE DE CONOCIMIENTO ES CONSISTENTE. Se encontraron {len(self.modelos_encontrados)} mundos posibles:\n")
            for idx, mod in enumerate(self.modelos_encontrados, 1):
                asignaciones = []
                for a in mod['afirmados']: asignaciones.append(f"{a} = V")
                for n in mod['negados']: asignaciones.append(f"{n} = F")
                print(f"Mundo {idx}: [ {', '.join(asignaciones)} ]")
        print("============================================")
